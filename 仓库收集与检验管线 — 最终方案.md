# 仓库收集与检验管线 — 最终方案

## 当前生效的采样与语言口径

- `500` 表示 GitHub 原始样本总数。当前通过五条语言查询各抽取 100 条原始记录，
  这只是构成固定规模样本的采样方式，不是各语言的通过配额。
- 每个仓库按相同的硬过滤、软评分、方向核查和 E2B 验证规则自然流转；某种语言
  通过多少就进入下一阶段多少，不补齐、不截断，也不因语言占比淘汰。
- S5 仅保留为历史兼容字段，默认固定记 2 分。只有显式设置
  `PIPELINE_LANGUAGE_QUOTA_ENABLED=true` 才启用下文的旧配额惩罚逻辑；本轮 500
  仓库运行未启用该开关。

## 筛选标准

### 硬性（全部满足才进入下一阶段）



|编号|标准|检测方式|
|---|---|---|
|H1|Stars ≥ 100|GitHub API `stargazers_count`|
|H2|近 12 个月有 commit|`pushed_at` 字段|
|H3|许可证为 MIT / Apache\-2\.0 / BSD\-2\-Clause / BSD\-3\-Clause / ISC / MPL\-2\.0|`license.spdx_id`|
|H4|有原生测试设施|检测 `pytest.ini`/`pyproject.toml`\(含pytest\)；或 `go.mod`\+至少一个`_test.go`；或 `package.json`\+jest/vitest；或 `Cargo.toml`\+`[dev-dependencies]`|
|H5|依赖可离线安装|`docker build --network=none` 成功退出|
|H6|目标方向无公开实现|GitHub code search API \+ grep\.app 关键词检索，0 结果才通过|



### 软性（满分 10，≥ 6 分才进入 Stage 4）



|编号|标准|分值逻辑|
|---|---|---|
|S1|文件数 200–3000|200–3000 → 2分；50–200 或 3000–8000 → 1分；其余 → 0分|
|S2|Stars 1k–50k|1k–50k → 2分；100–1k → 1分|
|S3|有 open issue 标注 enhancement/feature/roadmap|有 → 2分；无 → 0分|
|S4|exported public symbol 数 ≥ 20|满足 → 2分；否则 → 1分|
|S5|语言兼容字段（默认不启用配额）|默认固定 2 分；仅兼容模式按旧配额计算|



S5 说明：当前运行不根据语言占比筛除仓库。下文计数器只用于显式开启的兼容模式，
不影响默认管线或本轮结果。

## 管线实现



### Stage 0：候选抓取



```Python
import requests, os

HEADERS = {"Authorization": f"Bearer {os.environ['GITHUB_TOKEN']}",
           "Accept": "application/vnd.github+json"}

QUERIES = [
    "language:go stars:100..200000 pushed:>2024-01-01",
    "language:python stars:100..200000 pushed:>2024-01-01",
    "language:typescript stars:100..200000 pushed:>2024-01-01",
    "language:javascript stars:100..200000 pushed:>2024-01-01",
    "language:rust stars:100..200000 pushed:>2024-01-01",
]

def fetch_candidates(query: str, per_page: int = 100) -> list[dict]:
    r = requests.get("https://api.github.com/search/repositories",
                     headers=HEADERS,
                     params={"q": query, "sort": "updated", "order": "desc", "per_page": per_page})
    r.raise_for_status()
    return r.json()["items"]
```



### Stage 1：硬性过滤



```Python
from datetime import datetime, timezone

PERMISSIVE = {"MIT", "Apache-2.0", "BSD-2-Clause", "BSD-3-Clause", "ISC", "MPL-2.0"}

TEST_MARKERS = {
    "go":         ["go.mod"],
    "python":     ["pytest.ini", "pyproject.toml", "setup.cfg"],
    "typescript": ["package.json"],
    "javascript": ["package.json"],
    "rust":       ["Cargo.toml"],
}

def hard_filter(repo: dict) -> tuple[bool, str]:
    if repo["stargazers_count"] < 100:
        return False, "stars<100"
    pushed = datetime.fromisoformat(repo["pushed_at"].replace("Z", "+00:00"))
    if (datetime.now(timezone.utc) - pushed).days > 365:
        return False, "inactive"
    if not repo.get("license") or repo["license"]["spdx_id"] not in PERMISSIVE:
        return False, "license"
    lang = (repo.get("language") or "").lower()
    if not _has_any_file(repo["full_name"], TEST_MARKERS.get(lang, [])):
        return False, "no_test_infra"
    return True, "ok"

def _has_any_file(full_name: str, filenames: list[str]) -> bool:
    for f in filenames:
        r = requests.get(f"https://api.github.com/repos/{full_name}/contents/{f}",
                         headers=HEADERS)
        if r.status_code == 200:
            return True
    return False
```



### Stage 2：软性评分（默认关闭语言配额惩罚）



```Python
from collections import Counter

# 全局计数器：跨所有已入库仓库，按仓库数量比例控制语言分布
_lang_counter: Counter = Counter()

# 默认 False；本轮固定为 False。仅在复现实验性旧方案时显式开启。
ENFORCE_LANGUAGE_QUOTA = False

LANG_QUOTA = {
    "go": 0.40, "python": 0.40, "typescript": 0.40,
    "javascript": 0.15, "rust": 0.15,
}

def quota_ok(lang: str) -> bool:
    total = sum(_lang_counter.values()) + 1  # +1 预计入库当前仓库
    lang_count = _lang_counter[lang.lower()] + 1
    limit = LANG_QUOTA.get(lang.lower(), 0.10)
    return (lang_count / total) <= limit

def register_repo(lang: str):
    """入库时调用，更新全局计数器"""
    _lang_counter[lang.lower()] += 1

def estimate_file_count(full_name: str, sha: str = "HEAD") -> int:
    r = requests.get(f"https://api.github.com/repos/{full_name}/git/trees/{sha}",
                     headers=HEADERS, params={"recursive": "1"})
    r.raise_for_status()
    return sum(1 for item in r.json()["tree"] if item["type"] == "blob")

def has_feature_issues(full_name: str) -> bool:
    for label in ("enhancement", "feature", "roadmap"):
        r = requests.get(f"https://api.github.com/repos/{full_name}/issues",
                         headers=HEADERS,
                         params={"labels": label, "state": "open", "per_page": 1})
        if r.status_code == 200 and r.json():
            return True
    return False

def estimate_public_symbols(full_name: str, lang: str) -> int:
    """clone 后用简单正则统计 exported 符号"""
    patterns = {
        "go":         r"^func [A-Z]|^type [A-Z]",
        "python":     r"^def [a-z]|^class [A-Z]",
        "typescript": r"^export ",
        "javascript": r"^export |module\.exports",
        "rust":       r"^pub fn |^pub struct |^pub enum |^pub trait ",
    }
    return grep_count(full_name, patterns.get(lang.lower(), r"$^"))

def soft_score(repo: dict) -> float:
    score = 0
    lang = (repo.get("language") or "").lower()
    fc = estimate_file_count(repo["full_name"])
    if 200 <= fc <= 3000:                       score += 2
    elif 50 <= fc < 200 or 3000 < fc <= 8000:   score += 1
    s = repo["stargazers_count"]
    if 1000 <= s <= 50000:  score += 2
    elif s >= 100:          score += 1
    if has_feature_issues(repo["full_name"]):   score += 2
    if estimate_public_symbols(repo["full_name"], lang) >= 20: score += 2
    else:                                        score += 1
    if not ENFORCE_LANGUAGE_QUOTA or quota_ok(lang): score += 2
    return score
```



### Stage 3：方向可行性核查（全自动）



```Python
def check_direction(repo: dict) -> dict | None:
    """从 open issues 提取方向 → LLM 判定未实现 → 全网核查无公开实现"""
    issues = fetch_feature_issues(repo["full_name"], limit=10)
    for issue in issues:
        # 1. LLM 判定：该功能在当前代码库中是否尚未实现
        verdict = llm_judge(
            system="你是代码库分析员。判断 issue 描述的功能是否已在仓库中实现。",
            user=f"仓库结构:\n{get_repo_tree_summary(repo)}\n\nIssue:\n{issue['title']}\n{issue['body'][:3000]}",
            output_schema={"implemented": "bool", "behavior_boundary_clear": "bool",
                           "estimated_loc": "int", "keywords": "list[str]"}
        )
        if verdict["implemented"] or not verdict["behavior_boundary_clear"]:
            continue
        if verdict["estimated_loc"] < 200:   # 任务太浅
            continue
        # 2. H6：全网核查无公开实现
        if github_code_search(verdict["keywords"]) > 0:
            continue
        if grep_app_search(verdict["keywords"]) > 0:
            continue
        return {"source": f"issue#{issue['number']}",
                "direction": llm_summarize_direction(issue),
                "keywords": verdict["keywords"]}
    return None  # 无可用方向 → 该仓库标记 no_direction，跳过
```



### Stage 4：离线环境构建验证（H5）



```Python
import subprocess, tempfile

DOCKERFILE_TEMPLATES = {
    "go":     "FROM golang:1.22\nWORKDIR /repo\nCOPY . .\nRUN go mod download && go build ./...",
    "python": "FROM python:3.11\nWORKDIR /repo\nCOPY . .\nRUN pip install -e '.[test,dev]' || pip install -e .",
    "typescript": "FROM node:20\nWORKDIR /repo\nCOPY . .\nRUN npm ci",
    "javascript": "FROM node:20\nWORKDIR /repo\nCOPY . .\nRUN npm ci",
    "rust":   "FROM rust:1.77\nWORKDIR /repo\nCOPY . .\nRUN cargo build --tests",
}

TEST_CMDS = {
    "go": "go test ./...", "python": "python -m pytest -x -q",
    "typescript": "npx jest --ci", "javascript": "npx jest --ci",
    "rust": "cargo test",
}

def build_offline(repo: dict, base_commit: str) -> dict:
    lang = repo["language"].lower()
    with tempfile.TemporaryDirectory() as ws:
        subprocess.run(["git", "clone", "--depth=1",
                        f"https://github.com/{repo['full_name']}", ws], check=True)
        subprocess.run(["git", "checkout", base_commit], cwd=ws, check=True)
        with open(f"{ws}/Dockerfile.pipeline", "w") as f:
            f.write(DOCKERFILE_TEMPLATES[lang])
        tag = f"repo-env:{repo['full_name'].replace('/', '-')}"
        # 第一次联网 build（拉依赖并缓存进镜像层）
        r1 = subprocess.run(["docker", "build", "-f", "Dockerfile.pipeline",
                             "-t", tag, "."], cwd=ws, capture_output=True, timeout=600)
        if r1.returncode != 0:
            return {"ok": False, "reason": "build_fail"}
        # 离线验证：断网容器内跑测试，确认无运行时联网依赖
        r2 = subprocess.run(["docker", "run", "--rm", "--network=none", tag,
                             "sh", "-c", TEST_CMDS[lang]],
                            capture_output=True, timeout=600)
        return {"ok": r2.returncode == 0, "image": tag,
                "reason": "ok" if r2.returncode == 0 else "offline_test_fail"}
```



**构建耗时阈值：10 分钟**（`timeout=600`），超时判 `build_timeout` 排除。



### Stage 5：e2b 执行开销基准测试



先把 Stage 4 产出的镜像注册为 e2b template（`e2b template build`），再跑基准：



```Python
import time
from e2b import Sandbox

THRESHOLDS = {
    "cold_start_s": 20,       # 沙箱启动 < 20s
    "test_duration_s": 120,   # 测试套件 < 120s
    "peak_mem_mb": 4096,      # 峰值内存 < 4GB
}

def benchmark(template_id: str, test_cmd: str, n_runs: int = 3) -> dict:
    runs = []
    for _ in range(n_runs):
        t0 = time.monotonic()
        with Sandbox(template=template_id) as sbx:
            cold_start = time.monotonic() - t0
            # 用 /usr/bin/time -v 采集峰值内存(Maximum resident set size, KB)
            t1 = time.monotonic()
            proc = sbx.process.start_and_wait(
                f"/usr/bin/time -v sh -c '{test_cmd}' 2> /tmp/time.log")
            duration = time.monotonic() - t1
            mem_log = sbx.filesystem.read("/tmp/time.log")
            peak_kb = _parse_max_rss(mem_log)   # 提取 "Maximum resident set size"
            runs.append({"cold_start_s": cold_start, "test_duration_s": duration,
                         "exit_code": proc.exit_code, "peak_mem_mb": peak_kb / 1024})

    med = lambda k: sorted(r[k] for r in runs)[len(runs) // 2]
    durs = sorted(r["test_duration_s"] for r in runs)
    result = {
        "cold_start_median_s": round(med("cold_start_s"), 2),
        "test_duration_median_s": round(med("test_duration_s"), 2),
        "peak_mem_median_mb": round(med("peak_mem_mb"), 1),
        "all_passed": all(r["exit_code"] == 0 for r in runs),
        "stable": durs[-1] - durs[0] < 15,   # 3次极差<15s
    }
    result["pass"] = (result["all_passed"] and result["stable"]
                      and result["cold_start_median_s"] < THRESHOLDS["cold_start_s"]
                      and result["test_duration_median_s"] < THRESHOLDS["test_duration_s"]
                      and result["peak_mem_median_mb"] < THRESHOLDS["peak_mem_mb"])
    return result

def _parse_max_rss(log: str) -> int:
    for line in log.splitlines():
        if "Maximum resident set size" in line:
            return int(line.split(":")[-1].strip())
    return -1
```



**判定与降级规则**：

- `cold_start` / `test_duration` / `peak_mem` 任一超阈值 → 尝试缩小测试范围（只跑与目标方向相关的包/模块子集，如 `go test ./pkg/target/...`），用子集重新基准；子集仍超 → 排除；

- `stable=False`（3次极差 ≥15s 或 exit\_code 不一致）→ 标记 `flaky_test_suite`，软性扣 2 分后若总分仍 ≥6 保留，否则排除。

## 主流程编排



```Python
def pipeline():
    for query in QUERIES:
        for repo in fetch_candidates(query):
            ok, reason = hard_filter(repo)                      # Stage 1
            if not ok: log_reject(repo, reason); continue
            score = soft_score(repo)                            # Stage 2
            if score < 6: log_reject(repo, f"score={score}"); continue
            direction = check_direction(repo)                   # Stage 3
            if not direction: log_reject(repo, "no_direction"); continue
            repo["base_commit"] = get_head_sha(repo)
            build = build_offline(repo, repo["base_commit"])    # Stage 4
            if not build["ok"]: log_reject(repo, build["reason"]); continue
            template_id = register_e2b_template(build["image"])
            bench = benchmark(template_id, TEST_CMDS[repo["language"].lower()])  # Stage 5
            if not bench["pass"]: log_reject(repo, "benchmark_fail"); continue
            register(repo, direction, build, bench, score)      # Stage 6
```



### 新增 S6：开发者库偏好（软性，2分）



```Python
DEVELOPER_LIB_SIGNALS = [
    # 正向信号：典型开发者库特征
    "library", "sdk", "framework", "parser", "serializer", "client",
    "driver", "codec", "protocol", "compiler", "linter", "formatter",
    "middleware", "router", "orm", "rpc", "cli", "toolkit",
    # 负向信号：应用/服务类，不是库
    "app", "server", "dashboard", "website", "blog", "game",
    "bot", "scraper", "crawler", "monitor", "deploy",
]

def is_developer_lib(repo: dict) -> bool:
    text = f"{repo['name']} {repo.get('description') or ''}".lower()
    topics = [t.lower() for t in repo.get("topics", [])]
    positive = sum(1 for s in DEVELOPER_LIB_SIGNALS[:18] if s in text or s in topics)
    negative = sum(1 for s in DEVELOPER_LIB_SIGNALS[18:] if s in text or s in topics)
    return positive > 0 and negative == 0
```

### 偏好约定说明（给 codex）



- S6 是软性分，不是硬性排除——应用类仓库若其他维度极强（如 stars 极高、有清晰 API 边界）仍可入库，只是得分偏低；
